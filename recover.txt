Here’s a minimal Rust sketch that takes a slice containing exactly one OSM PBF **blob** (one “chunk” in the PBF sense) and decodes the `PrimitiveBlock` from it, using the low‑level `osm-pbf`/`osmpbfreader`-style API.

I’ll assume:

- Your “chunk” starts at the 4‑byte length prefix of the `BlobHeader`.
- You have prost‑generated types from `fileformat.proto` and `osmformat.proto` (or use an existing crate that exposes them similarly).

```rust
use std::io::Cursor;

// Adjust imports to your crate:
// typically something like:
//
// use osm_pbf::fileformat::{BlobHeader, Blob};
// use osm_pbf::osmformat::PrimitiveBlock;
//
// or from osmpbfreader's re-exports:
// use osmpbfreader::fileformat::{BlobHeader, Blob};
// use osmpbfreader::osmformat::PrimitiveBlock;

fn read_single_pbf_chunk(chunk: &[u8]) -> anyhow::Result<PrimitiveBlock> {
    let mut cur = Cursor::new(chunk);

    // 1. Read 4-byte big-endian header length
    let header_len = {
        use std::io::Read;
        let mut buf = [0u8; 4];
        cur.read_exact(&mut buf)?;
        u32::from_be_bytes(buf) as usize
    };

    // 2. Read and decode BlobHeader
    let header_bytes = {
        use std::io::Read;
        let mut buf = vec![0u8; header_len];
        cur.read_exact(&mut buf)?;
        buf
    };

    let header: BlobHeader = prost::Message::decode(&header_bytes[..])?;

    // 3. Read Blob according to header.datasize
    let data_size = header.datasize as usize;
    let blob_bytes = {
        use std::io::Read;
        let mut buf = vec![0u8; data_size];
        cur.read_exact(&mut buf)?;
        buf
    };

    let blob: Blob = prost::Message::decode(&blob_bytes[..])?;

    // 4. Decompress blob into raw PrimitiveBlock bytes
    let raw = if let Some(zlib_data) = &blob.zlib_data {
        // zlib stream
        use flate2::read::ZlibDecoder;
        use std::io::Read;

        let mut decoder = ZlibDecoder::new(&zlib_data[..]);
        let mut out = Vec::with_capacity(blob.raw_size as usize);
        decoder.read_to_end(&mut out)?;
        out
    } else if let Some(raw) = &blob.raw {
        raw.clone()
    } else {
        // handle other compression methods (lz4, etc.) if needed
        anyhow::bail!("unsupported compression method in blob");
    };

    // 5. Decode PrimitiveBlock
    let primitive_block: PrimitiveBlock = prost::Message::decode(&raw[..])?;
    Ok(primitive_block)
}
```
